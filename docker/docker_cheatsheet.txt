# Docker cheat-sheet 

# build a Docker image 
# -t stands for tag.
docker build --no-cache -t didactic-stats-app .
docker build -t didactic-stats-app .

# Creates a new container from an image and starts it
# -p HOST_PORT:CONTAINER_PORT
# -d = detached: Docker runs it independently of terminal

docker run -d -p 3838:3838 --name app01 didactic-stats-app
docker run -d -p 3839:3838 --name app02 didactic-stats-app

# in browser 
http://localhost:3838/didactic_stats_app/
http://localhost:3839/didactic_stats_app/

# remove existing container 
docker rm -f app01

# erase image !!!
# docker rmi didactic-stats-app

# Start / stop an existing container.
docker start didactic-stats-test
docker stop didactic-stats-test
docker stop app01

# get info 
docker ps
docker ps -a

docker images
